using System.Data;
using Microsoft.Extensions.Logging;
using Oracle.ManagedDataAccess.Client;
using ProjectName.Repositories.IRepository;

namespace ProjectName.Repositories;

public class PortfolioRepository : IPortfolioRepository
{
    private readonly IOracleConnectionFactory _connectionFactory;
    private readonly ILogger<PortfolioRepository> _logger;

    public PortfolioRepository(IOracleConnectionFactory connectionFactory, ILogger<PortfolioRepository> logger)
    {
        _connectionFactory = connectionFactory;
        _logger = logger;
    }

    /// <summary>
    /// Consolidated query for portfolio summary from v_portfolio_all view.
    /// Replaces: GetUrGainAsync, GetUrLossAsync, GetEquityAssetValueAllAsync,
    ///           GetEquityAssetValueMarginableAsync, GetAssetValueAsync (5 queries -> 1)
    /// </summary>
    public async Task<DataTable?> GetPortfolioSummaryAsync(int accountNo, CancellationToken cancellationToken = default)
    {
        const string sql = @"
            SELECT
                NVL(SUM(CASE WHEN ur_gain > 0 THEN ur_gain ELSE 0 END), 0) AS urgain,
                NVL(SUM(CASE WHEN ur_gain < 0 THEN ur_gain ELSE 0 END), 0) AS urloss,
                NVL(SUM(m_val), 0) AS asset_val_all,
                NVL(SUM(CASE WHEN margin_nonmargin = 'M' THEN m_val ELSE 0 END), 0) AS asset_val_margin
            FROM invest.v_portfolio_all
            WHERE accnt_no = :accnt_no";

        return await ExecuteQueryAsync(sql, new[] { new OracleParameter("accnt_no", accountNo) }, cancellationToken);
    }

    /// <summary>
    /// Consolidated query for transaction orders.
    /// Replaces: GetFundWithdrawalAsync, GetFundWithdrawalOrderAsync, GetCollectionInTransitAsync (3 queries -> 1)
    /// </summary>
    public async Task<DataTable?> GetTransactionSummaryAsync(int accountNo, CancellationToken cancellationToken = default)
    {
        const string sql = @"
            SELECT
                NVL(SUM(CASE WHEN tr_cd = 'WF' AND dr_cr = 'D' THEN amount ELSE 0 END), 0) AS fund_withdraw,
                NVL(SUM(CASE WHEN tr_cd = 'AD' THEN amount ELSE 0 END), 0) AS collection_in_transit
            FROM invest.trans_ord
            WHERE accnt_no = :accnt_no";

        return await ExecuteQueryAsync(sql, new[] { new OracleParameter("accnt_no", accountNo) }, cancellationToken);
    }

    /// <summary>
    /// Consolidated query for account balance from dprofile.
    /// Replaces: GetAccruedInterestAsync, GetCFmbalAsync (2 queries -> 1)
    /// </summary>
    public async Task<DataTable?> GetAccountBalanceAsync(int accountNo, CancellationToken cancellationToken = default)
    {
        const string sql = @"
            SELECT
                NVL(interest, 0) AS accured_interest,
                NVL(cur_bal, 0) AS mbal
            FROM invest.dprofile
            WHERE accnt_no = :accnt_no";

        return await ExecuteQueryAsync(sql, new[] { new OracleParameter("accnt_no", accountNo) }, cancellationToken);
    }

    /// <summary>
    /// Consolidated query for financial view data.
    /// Replaces: GetAppliedInterestAsync, GetEarnedDivIncomeAsync (2 queries -> 1)
    /// </summary>
    public async Task<DataTable?> GetFinancialSummaryAsync(int accountNo, CancellationToken cancellationToken = default)
    {
        const string sql = @"
            SELECT
                ABS(NVL(SUM(CASE WHEN tr_cd IN ('INT', 'DMT', 'CFS', 'CSTD') AND dr_cr = 'D' AND vouch_dt >= DATE '2022-07-01' THEN amount ELSE 0 END), 0)) AS applied_interest,
                NVL(SUM(CASE WHEN tr_cd IN ('FDIV', 'IDIV', 'BNS') AND dr_cr = 'C' AND vouch_dt >= DATE '2022-07-01' THEN amount ELSE 0 END), 0) AS earned_div_income
            FROM invest.fin_view
            WHERE accnt_no = :accnt_no";

        return await ExecuteQueryAsync(sql, new[] { new OracleParameter("accnt_no", accountNo) }, cancellationToken);
    }

    private async Task<DataTable?> ExecuteQueryAsync(string sql, OracleParameter[] parameters, CancellationToken cancellationToken)
    {
        var dt = new DataTable();
        try
        {
            using var conn = _connectionFactory.CreateConnection();
            await conn.OpenAsync(cancellationToken);

            using var cmd = new OracleCommand(sql, conn);
            cmd.BindByName = true;
            cmd.Parameters.AddRange(parameters);

            using var adapter = new OracleDataAdapter(cmd);
            await Task.Run(() => adapter.Fill(dt), cancellationToken);

            return dt.Rows.Count > 0 ? dt : null;
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error executing query");
            throw;
        }
    }
}

using System.Text.RegularExpressions;

namespace ProjectName.Services;

public static partial class EmailValidator
{
    [GeneratedRegex(@"^[^@\s]+@[^@\s]+\.[^@\s]+$", RegexOptions.IgnoreCase, matchTimeoutMilliseconds: 200)]
    private static partial Regex EmailRegex();

    public static bool IsValidEmail(string? email)
    {
        if (string.IsNullOrWhiteSpace(email))
            return false;
        try
        {
            return EmailRegex().IsMatch(email.Trim());
        }
        catch (RegexMatchTimeoutException)
        {
            return false;
        }
    }
}

--- MailKitEmailService.cs (with Polly retry) ---
using MailKit.Net.Smtp;
using MailKit.Security;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.Options;
using MimeKit;
using ProjectName.Repositories.IRepository;
using ProjectName.Configuration;
using Polly;
using Polly.Retry;

namespace ProjectName.Services;

public class MailKitEmailService : IEmailService
{
    private readonly EmailSettings _emailSettings;
    private readonly RetryPolicySettings _retrySettings;
    private readonly ILogger<MailKitEmailService> _logger;
    private readonly ResiliencePipeline _retryPipeline;

    public MailKitEmailService(
        IOptions<EmailSettings> emailSettings,
        IOptions<RetryPolicySettings> retrySettings,
        ILogger<MailKitEmailService> logger)
    {
        _emailSettings = emailSettings.Value;
        _retrySettings = retrySettings.Value;
        _logger = logger;
        _retryPipeline = BuildRetryPipeline();
    }

    private ResiliencePipeline BuildRetryPipeline()
    {
        return new ResiliencePipelineBuilder()
            .AddRetry(new RetryStrategyOptions
            {
                MaxRetryAttempts = _retrySettings.MaxRetryAttempts,
                Delay = TimeSpan.FromSeconds(_retrySettings.InitialDelaySeconds),
                MaxDelay = TimeSpan.FromSeconds(_retrySettings.MaxDelaySeconds),
                BackoffType = DelayBackoffType.Exponential,
                UseJitter = true,
                ShouldHandle = new PredicateBuilder()
                    .Handle<SmtpCommandException>()
                    .Handle<SmtpProtocolException>()
                    .Handle<IOException>()
                    .Handle<TimeoutException>(),
                OnRetry = args =>
                {
                    _logger.LogWarning(args.Outcome.Exception,
                        "Email attempt {Attempt} failed. Retrying...",
                        args.AttemptNumber + 1);
                    return ValueTask.CompletedTask;
                }
            })
            .Build();
    }

    public async Task<EmailResult> SendEmailAsync(
        IEnumerable<(byte[] Content, string FileName)> attachments,
        string toAddress, string subject, string body,
        CancellationToken cancellationToken = default)
    {
        if (!EmailValidator.IsValidEmail(toAddress))
            return new EmailResult(false, "Invalid email address");

        try
        {
            await _retryPipeline.ExecuteAsync(async token =>
            {
                await SendEmailInternalAsync(attachments, toAddress, subject, body, token);
            }, cancellationToken);

            return new EmailResult(true, "Success");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Failed to send email to {Email}", toAddress);
            return new EmailResult(false, $"Failure: {ex.Message}");
        }
    }

    private async Task SendEmailInternalAsync(
        IEnumerable<(byte[] Content, string FileName)> attachments,
        string toAddress, string subject, string body,
        CancellationToken cancellationToken)
    {
        var message = new MimeMessage();
        message.From.Add(MailboxAddress.Parse(_emailSettings.FromAddress));
        message.To.Add(MailboxAddress.Parse(toAddress.Trim()));
        message.Subject = subject;

        var builder = new BodyBuilder { HtmlBody = body };
        foreach (var (content, fileName) in attachments)
        {
            if (content?.Length > 0)
                builder.Attachments.Add(fileName, content, new ContentType("application", "pdf"));
        }
        message.Body = builder.ToMessageBody();

        using var client = new SmtpClient();
        var secureSocketOptions = _emailSettings.EnableSsl
            ? SecureSocketOptions.StartTls : SecureSocketOptions.None;

        await client.ConnectAsync(_emailSettings.SmtpServer, _emailSettings.Port,
            secureSocketOptions, cancellationToken);

        if (!_emailSettings.UseDefaultCredentials && !string.IsNullOrEmpty(_emailSettings.Username))
            await client.AuthenticateAsync(_emailSettings.Username, _emailSettings.Password, cancellationToken);

        await client.SendAsync(message, cancellationToken);
        await client.DisconnectAsync(true, cancellationToken);
    }
}

--- EmailWorkerService.cs (BackgroundService) ---
using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;
using ProjectName.Repositories.IRepository;

namespace ProjectName.Services;

public class EmailWorkerService : BackgroundService
{
    private readonly IServiceScopeFactory _scopeFactory;
    private readonly IHostApplicationLifetime _applicationLifetime;
    private readonly ILogger<EmailWorkerService> _logger;

    public EmailWorkerService(
        IServiceScopeFactory scopeFactory,
        IHostApplicationLifetime applicationLifetime,
        ILogger<EmailWorkerService> logger)
    {
        _scopeFactory = scopeFactory;
        _applicationLifetime = applicationLifetime;
        _logger = logger;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _logger.LogInformation("Service starting at {Time}", DateTimeOffset.Now);

        try
        {
            using var scope = _scopeFactory.CreateScope();
            var workflow = scope.ServiceProvider.GetRequiredService<IEmailWorkflowService>();
            await workflow.ExecuteAsync(stoppingToken);
            _logger.LogInformation("Service completed at {Time}", DateTimeOffset.Now);
        }
        catch (OperationCanceledException)
        {
            _logger.LogWarning("Service was cancelled");
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Service failed");
        }
        finally
        {
            _logger.LogInformation("Service shutting down");
            _applicationLifetime.StopApplication();
        }
    }
}

using Microsoft.Extensions.DependencyInjection;
using Microsoft.Extensions.Hosting;
using ProjectName.Configuration;
using ProjectName.Data;
using ProjectName.Repositories;
using ProjectName.Repositories.IRepository;
using ProjectName.Services;
using Serilog;

// Bootstrap logger
Log.Logger = new LoggerConfiguration()
    .WriteTo.Console()
    .CreateBootstrapLogger();

try
{
    Log.Information("Starting service");

    var builder = Host.CreateApplicationBuilder(args);

    // Configure Serilog
    builder.Services.AddSerilog((services, lc) => lc
        .ReadFrom.Configuration(builder.Configuration)
        .ReadFrom.Services(services)
        .Enrich.FromLogContext()
        .Enrich.WithMachineName()
        .Enrich.WithThreadId()
        .WriteTo.Console(outputTemplate: "[{Timestamp:HH:mm:ss} {Level:u3}] {Message:lj}{NewLine}{Exception}")
        .WriteTo.File(
            path: "logs/service-.log",
            rollingInterval: RollingInterval.Day,
            outputTemplate: "{Timestamp:yyyy-MM-dd HH:mm:ss.fff zzz} [{Level:u3}] [{ThreadId}] {Message:lj}{NewLine}{Exception}"));

    // Configure options
    builder.Services.Configure<EmailSettings>(builder.Configuration.GetSection("Email"));
    builder.Services.Configure<RetryPolicySettings>(builder.Configuration.GetSection("RetryPolicy"));
    builder.Services.Configure<ApplicationSettings>(builder.Configuration.GetSection("Application"));

    // Register data access
    builder.Services.AddSingleton<IOracleConnectionFactory, OracleConnectionFactory>();

    // Register repositories
    builder.Services.AddScoped<IPortfolioRepository, PortfolioRepository>();

    // Register services
    builder.Services.AddScoped<IEmailService, MailKitEmailService>();
    builder.Services.AddScoped<IPdfGenerationService, PdfGenerationService>();
    builder.Services.AddScoped<IEmailWorkflowService, EmailWorkflowService>();

    // Register hosted service
    builder.Services.AddHostedService<EmailWorkerService>();

    var host = builder.Build();
    await host.RunAsync();
}
catch (Exception ex)
{
    Log.Fatal(ex, "Service terminated unexpectedly");
}
finally
{
    Log.CloseAndFlush();
}

14. SQL QUERY OPTIMIZATION
================================================================================

This section covers SQL optimization strategies to reduce database round trips
and improve application performance.

--------------------------------------------------------------------------------
14.1 QUERY CONSOLIDATION STRATEGY
--------------------------------------------------------------------------------

PROBLEM: Multiple queries hitting the same table/view

BEFORE (16 separate queries per account):
- GetUrGainAsync         -> SELECT SUM(ur_gain) FROM v_portfolio_all WHERE ur_gain > 0
- GetUrLossAsync         -> SELECT SUM(ur_gain) FROM v_portfolio_all WHERE ur_gain < 0
- GetAssetValueAllAsync  -> SELECT SUM(m_val) FROM v_portfolio_all
- GetAssetValMarginAsync -> SELECT SUM(m_val) FROM v_portfolio_all WHERE margin='M'
- GetFundWithdrawalAsync -> SELECT SUM(amount) FROM trans_ord WHERE tr_cd='WF'
- GetCollectionAsync     -> SELECT SUM(amount) FROM trans_ord WHERE tr_cd='AD'
- GetAccruedInterest     -> SELECT interest FROM dprofile
- GetCurBalance          -> SELECT cur_bal FROM dprofile
- GetAppliedInterest     -> SELECT SUM(amount) FROM fin_view WHERE tr_cd IN (...)
- GetEarnedDivIncome     -> SELECT SUM(amount) FROM fin_view WHERE tr_cd IN (...)
... and more

AFTER (10 consolidated queries per account):

1. GetPortfolioSummaryAsync (5 queries -> 1)
   Combines: urgain, urloss, asset_val_all, asset_val_margin

2. GetTransactionSummaryAsync (3 queries -> 1)
   Combines: fund_withdraw, collection_in_transit

3. GetAccountBalanceAsync (2 queries -> 1)
   Combines: accured_interest, mbal

4. GetFinancialSummaryAsync (2 queries -> 1)
   Combines: applied_interest, earned_div_income

RESULT: 8 database round-trips eliminated per account

--------------------------------------------------------------------------------
14.2 CONSOLIDATION PATTERNS
--------------------------------------------------------------------------------

PATTERN 1: Use CASE statements to combine aggregations

BEFORE (2 queries):
SELECT SUM(ur_gain) FROM view WHERE ur_gain > 0
SELECT SUM(ur_gain) FROM view WHERE ur_gain < 0

AFTER (1 query):
SELECT
    NVL(SUM(CASE WHEN ur_gain > 0 THEN ur_gain ELSE 0 END), 0) AS urgain,
    NVL(SUM(CASE WHEN ur_gain < 0 THEN ur_gain ELSE 0 END), 0) AS urloss
FROM view WHERE accnt_no = :accnt_no

PATTERN 2: Combine same-table queries with different filters

BEFORE (3 queries):
SELECT SUM(amount) FROM trans_ord WHERE tr_cd = 'WF' AND dr_cr = 'D'
SELECT SUM(amount) FROM trans_ord WHERE tr_cd = 'WF' AND dr_cr = 'D'  -- DUPLICATE!
SELECT SUM(amount) FROM trans_ord WHERE tr_cd = 'AD'

AFTER (1 query):
SELECT
    NVL(SUM(CASE WHEN tr_cd = 'WF' AND dr_cr = 'D' THEN amount ELSE 0 END), 0) AS fund_withdraw,
    NVL(SUM(CASE WHEN tr_cd = 'AD' THEN amount ELSE 0 END), 0) AS collection_in_transit
FROM invest.trans_ord WHERE accnt_no = :accnt_no

PATTERN 3: Select multiple columns instead of multiple queries

BEFORE (2 queries):
SELECT interest FROM dprofile WHERE accnt_no = :id
SELECT cur_bal FROM dprofile WHERE accnt_no = :id

AFTER (1 query):
SELECT
    NVL(interest, 0) AS accured_interest,
    NVL(cur_bal, 0) AS mbal
FROM invest.dprofile WHERE accnt_no = :accnt_no

--------------------------------------------------------------------------------
14.3 PARALLEL QUERY EXECUTION
--------------------------------------------------------------------------------

Execute independent queries in parallel using Task.WhenAll():

BEFORE (Sequential - slow):
var purchasePower = await _repository.GetPurchasePowerAsync(accountNo);
var loanRatio = await _repository.GetLoanRatioAsync();
var marginDeposit = await _repository.GetMarginDepositAsync(accountNo);
var portfolioSummary = await _repository.GetPortfolioSummaryAsync(accountNo);
// ... each query waits for the previous one to complete

AFTER (Parallel - fast):
var purchasePowerTask = _repository.GetPurchasePowerAsync(accountNo, cancellationToken);
var loanRatioTask = _repository.GetLoanRatioAsync(cancellationToken);
var marginDepositTask = _repository.GetMarginDepositAsync(accountNo, cancellationToken);
var portfolioSummaryTask = _repository.GetPortfolioSummaryAsync(accountNo, cancellationToken);
var transactionSummaryTask = _repository.GetTransactionSummaryAsync(accountNo, cancellationToken);
var accountBalanceTask = _repository.GetAccountBalanceAsync(accountNo, cancellationToken);
var financialSummaryTask = _repository.GetFinancialSummaryAsync(accountNo, cancellationToken);
var receivableFundTask = _repository.GetReceivableFundAsync(accountNo, cancellationToken);
var ipoBlockBalanceTask = _repository.GetIPOBlockBalanceAsync(accountNo, cancellationToken);
var receivableDividendTask = _repository.GetReceivableDividendAsync(accountNo, cancellationToken);

// Execute all queries in parallel
await Task.WhenAll(
    purchasePowerTask, loanRatioTask, marginDepositTask, portfolioSummaryTask,
    transactionSummaryTask, accountBalanceTask, financialSummaryTask,
    receivableFundTask, ipoBlockBalanceTask, receivableDividendTask
);

// Retrieve results (already completed)
var purchasePower = await purchasePowerTask;
var loanRatio = await loanRatioTask;
// ... etc

14.4 SQL BEST PRACTICES
--------------------------------------------------------------------------------

1. Use explicit JOIN syntax instead of comma-separated tables:

   BEFORE:
   SELECT * FROM table_a a, table_b b WHERE a.id = b.id

   AFTER:
   SELECT * FROM table_a a
   JOIN table_b b ON a.id = b.id

2. Use ANSI SQL DATE literals:

   BEFORE:
   WHERE vouch_dt >= '01-jul-2022'

   AFTER:
   WHERE vouch_dt >= DATE '2022-07-01'

3. Always use NVL/COALESCE for nullable aggregations:

   BEFORE:
   SELECT SUM(amount) FROM table

   AFTER:
   SELECT NVL(SUM(amount), 0) AS total_amount FROM table

4. Remove unused parameters from method signatures:

   BEFORE:
   Task<DataTable?> GetDataAsync(int branchCode, int accountNo)  // branchCode not used

   AFTER:
   Task<DataTable?> GetDataAsync(int accountNo)

5. Identify and remove duplicate queries:

   If two methods have identical SQL, consolidate them into one.

15. PERFORMANCE BEST PRACTICES
================================================================================

--------------------------------------------------------------------------------
15.1 DATABASE CONNECTION MANAGEMENT
--------------------------------------------------------------------------------

- Use connection factory pattern for consistent connection handling
- Always use 'using' statements for automatic disposal
- Don't hold connections open longer than necessary
- Use async methods (OpenAsync, ExecuteAsync)

--------------------------------------------------------------------------------
15.2 MEMORY OPTIMIZATION
--------------------------------------------------------------------------------

- Use Array.Empty<byte>() instead of new byte[0]
- Use StringBuilder for string concatenation in loops
- Dispose of large objects (DataTables, streams) promptly
- Use value types (struct) for small, immutable data

--------------------------------------------------------------------------------
15.3 ASYNC/AWAIT BEST PRACTICES
--------------------------------------------------------------------------------

- Always pass CancellationToken through the call chain
- Use ConfigureAwait(false) in library code
- Don't use async void (except for event handlers)
- Use Task.WhenAll for parallel operations

--------------------------------------------------------------------------------
15.4 LOGGING BEST PRACTICES
--------------------------------------------------------------------------------

- Use structured logging with named parameters
- Log at appropriate levels (Debug, Info, Warning, Error)
- Don't log sensitive data (passwords, connection strings)
- Include correlation IDs for tracing

--------------------------------------------------------------------------------
